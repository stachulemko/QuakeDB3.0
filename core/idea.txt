  Oto pseudokod funkcji sprawdzającej konflikty podczas operacji UPDATE i instrukcja jej obsługi:

  ### Pseudokod funkcji weryfikującej konflikt:
    // Zwracane statusy akcji do podjęcia przez silnik zapytań
    enum Action {
        PROCEED,                  // Możesz bezpiecznie aktualizować krotkę
        WAIT,                     // Transakcja blokująca wciąż trwa, musimy zawiesić działanie (dodać do TabOfBlockedTransaction)
        ERROR_CONCURRENT_UPDATE,  // BŁĄD RR: inna transakcja zmodyfikowała krotkę! Przerywamy naszą transakcję
        RE_EVALUATE_NEW_VERSION   // RC: inna transakcja zatwierdziła, musimy odszukać po łańcuchu (chain) nową wersję
    };

    Action check_update_conflict(Tuple *tuple, int current_txn_id, int isolation_level, MVCC *mvcc) {
        int blocking_xid = tuple->header.t_xmax;

        // 1. Krotka nie była przez nikogo zablokowana (xmax to 0)
        // lub blokuje ją nasza własna transakcja
        if (blocking_xid == 0 || blocking_xid == current_txn_id) {
            return PROCEED;
        }

        // 2. Ktoś inny zmodyfikował krotkę (lub zablokował przy SELECT FOR UPDATE)
        if (mvcc_is_txn_active(mvcc, blocking_xid)) {
            // INNA TRANSAKCJA WCIĄŻ DZIAŁA.
            // Niezależnie czy to RC, czy RR – musimy poczekać na jej koniec.
            return WAIT;
        } else {
            // INNA TRANSAKCJA SIĘ ZAKOŃCZYŁA. Sprawdzamy czy był COMMIT, czy ROLLBACK.
            if (mvcc_did_txn_commit(mvcc, blocking_xid)) {
                // BYŁ COMMIT (ktoś pomyślnie zaktualizował nasz rekord)

                if (isolation_level == REPEATABLE_READ) {
                    // W RR jest to krytyczny błąd współbieżności!
                    // Zrzucamy błąd i przerywamy current_txn_id.
                    return ERROR_CONCURRENT_UPDATE;

                } else if (isolation_level == READ_COMMITTED) {
                    // W RC przechodzimy na nowszą wersję. Silnik zapytania musi:
                    // a) podążyć po t_cid do nowej krotki,
                    // b) sprawdzić czy wciąż spełnia warunek WHERE,
                    // c) ponowić próbę UPDATE na nowej krotce.
                    return RE_EVALUATE_NEW_VERSION;
                }
            } else {
                // BYŁ ROLLBACK (inna transakcja wycofała swoje zmiany)
                // Jej blokada jest nieważna, możemy bezpiecznie nadpisać rekord.
                return PROCEED;
            }
        }
    }
